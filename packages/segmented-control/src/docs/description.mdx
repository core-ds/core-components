## Размеры

Компонент доступен в размерах 48, 40, 32.

```jsx live
render(() => {
    const [selectedId, setSelectedId] = React.useState({
        1: 1,
        2: 1,
        3: 1,
    });
    const handleChange = (pos, id) => setSelectedId({ ...selectedId, [pos]: id });
    return (
        <div style={{ width: 436 }}>
            <Space fullWidth size={12}>
                <SegmentedControl
                    size={48}
                    selectedId={selectedId['1']}
                    onChange={(id) => handleChange('1', id)}
                >
                    <Segment id={1} title='Size' />
                    <Segment id={2} title='48' />
                </SegmentedControl>
                <SegmentedControl
                    size={40}
                    selectedId={selectedId['2']}
                    onChange={(id) => handleChange('2', id)}
                >
                    <Segment id={1} title='Size' />
                    <Segment id={2} title='40' />
                </SegmentedControl>
                <SegmentedControl
                    size={32}
                    selectedId={selectedId['3']}
                    onChange={(id) => handleChange('3', id)}
                >
                    <Segment id={1} title='Size' />
                    <Segment id={2} title='32' />
                </SegmentedControl>
            </Space>
        </div>
    );
});
```

## Анатомия

С помощью слотов leftAddons и rightAddons можно кастомизировать тэг. Например, добавить иконку. Переданный контент будет отрисован слева или справа от текста тэга.

```jsx live
render(() => {
    const [selectedId, setSelectedId] = React.useState({
        1: 1,
    });
    const handleChange = (pos, id) => setSelectedId({ ...selectedId, [pos]: id });
    const diamonds20 = { content: <Diamonds20Icon /> };
    const diamonds24 = { content: <Diamonds24Icon /> };
    return (
        <div style={{ width: 310 }}>
            <SegmentedControl
                size={40}
                shape='rounded'
                segmentWidth='content'
                selectedId={selectedId['1']}
                onChange={(id) => handleChange('1', id)}
            >
                <Segment id={1} title={null} addons={{ left: diamonds24 }} />
                <Segment id={2} title='Left' addons={{ left: diamonds20 }} />
                <Segment id={3} title='Right' addons={{ right: diamonds20 }} />
                <Segment id={4} title='Both' addons={{ left: diamonds20, right: diamonds20 }} />
            </SegmentedControl>
        </div>
    );
});
//MOBILE
```
