## Примеры

Значения могут быть положительными и отрицательными.

```jsx live
render(() => {
    const [positiveOnly, setPositiveOnly] = React.useState(false);
    const [value, setValue] = React.useState(-123456767);
    const handlePositiveOnlyChange = (_, payload) => {
        setPositiveOnly(!payload.checked);
        setValue((prevValue) => prevValue * -1);
    };
    const handleValueChange = (_, payload) => {
        setValue(payload.value);
    };

    return (
        <React.Fragment>
            <div style={{ width: isMobile() ? '100%' : 320 }}>
                <AmountInput
                    value={value}
                    onChange={handleValueChange}
                    placeholder='Введите сумму'
                    labelView={isMobile() ? 'outer' : 'inner'}
                    label='Сумма'
                    minority={100}
                    bold={false}
                    block={true}
                    size={isMobile() ? 48 : 56}
                    breakpoint={BREAKPOINT}
                    positiveOnly={positiveOnly}
                />
            </div>
            <Gap size={20} />
            <Divider />
            <Gap size={20} />
            <Switch
                label='Принимать отрицательные значения'
                checked={!positiveOnly}
                onChange={handlePositiveOnlyChange}
            />
        </React.Fragment>
    );
});
```

Максимальное количество цифр в поле — 15, разделённых запятой в любом месте.

```jsx live
render(() => {
    const [value, setValue] = React.useState(123456789012345);
    const [minor, setMinor] = React.useState(2);
    const handleValueChange = (_, payload) => {
        setValue(payload.value);
    };
    const handleMinorChange = (_, payload) => {
        setMinor(payload.value);
    };

    return (
        <div style={{ width: isMobile() ? '100%' : 320 }}>
            <AmountInput
                value={value}
                onChange={handleValueChange}
                placeholder='Введите сумму'
                labelView={isMobile() ? 'outer' : 'inner'}
                label='Сумма'
                minority={Math.pow(10, minor)}
                integerLength={15 - minor}
                bold={false}
                block={true}
                size={isMobile() ? 48 : 56}
                breakpoint={BREAKPOINT}
            />
            <Gap size={20} />
            <NumberInput
                value={minor}
                onChange={handleMinorChange}
                step={1}
                min={0}
                max={14}
                block={true}
                disableUserInput={true}
                placeholder='Введите число знаков после зяпатой'
                label='Знаки после зяпатой'
                labelView={isMobile() ? 'outer' : 'inner'}
                size={isMobile() ? 48 : 56}
                breakpoint={BREAKPOINT}
            />
        </div>
    );
});
```

Для отображения валюты можно использовать суффикс.

```jsx live
render(() => {
    const [suffix, setSuffix] = React.useState('symbolic');
    const [value, setValue] = React.useState(1234567);
    const handleChange = (_, payload) => {
        setValue(payload.value);
    };
    const suffixInput = {
        percent: '%',
        noSuffix: null,
    }[suffix];
    const codeFormat = {
        symbolic: 'symbolic',
        letter: 'letter',
    }[suffix];
    const onSuffixChange = (_, payload) => {
        setSuffix(payload.value);
    };

    return (
        <React.Fragment>
            <div style={{ width: isMobile() ? '100%' : 320 }}>
                <AmountInput
                    value={value}
                    onChange={handleChange}
                    placeholder='Введите сумму'
                    labelView={isMobile() ? 'outer' : 'inner'}
                    label='Сумма'
                    minority={100}
                    bold={false}
                    currency='RUB'
                    suffix={suffixInput}
                    codeFormat={codeFormat}
                    block={true}
                    size={isMobile() ? 48 : 56}
                    breakpoint={BREAKPOINT}
                />
            </div>
            <Gap size={20} />
            <Divider />
            <Gap size={20} />
            <RadioGroup
                label='Суффикс'
                direction='vertical'
                name='radioGroup'
                onChange={onSuffixChange}
                value={suffix}
            >
                <Radio size={24} label='Знак ₽' value='symbolic' />
                <Radio size={24} label='Код RUB' value='letter' />
                <Radio size={24} label='Процент' value='percent' />
                <Radio size={24} label='Без суффикса' value='noSuffix' />
            </RadioGroup>
        </React.Fragment>
    );
});
```

Если знаечния изменяются с фиксированным шагом, можно включить степпер. При наличии степпера ввод с клавиатуры можно запретить.

```jsx live mobileHeight={500}
render(() => {
    const [disableUserInput, setDisableUserInput] = React.useState(false);
    const [value, setValue] = React.useState(1010);

    const handleChange = (e, payload) => {
        setValue(payload.value);
    };

    return (
        <React.Fragment>
            <div style={{ width: isMobile() ? '100%' : 320 }}>
                <AmountInput
                    value={value}
                    disableUserInput={disableUserInput}
                    placeholder='Введите сумму'
                    labelView={isMobile() ? 'outer' : 'inner'}
                    label='Сумма'
                    minority={100}
                    bold={false}
                    suffix='RUR'
                    block={true}
                    size={isMobile() ? 48 : 56}
                    breakpoint={BREAKPOINT}
                    onChange={handleChange}
                    stepper={{ step: 10, min: 1000, max: 1100 }}
                />
            </div>
            <Gap size={20} />
            <Divider />
            <Gap size={20} />
            <Switch
                block={true}
                checked={disableUserInput}
                label='Запретить ввод с клавиатуры'
                onChange={() => setDisableUserInput(!disableUserInput)}
            />
        </React.Fragment>
    );
});
```

Если инпут по умолчанию должен содержать ноль в качестве value, используйте `zeroValue={true}`, чтобы упростить ввод. Эта опция восстанавливает значение по умолчанию при очистке инпута и убирает ноль при получении фокуса.

```jsx live
render(() => {
    const [zeroValue, setZeroValue] = React.useState(true);
    const [value, setValue] = React.useState(0);
    const handleZeroValueChange = (_, payload) => {
        setZeroValue(payload.checked);
    };
    const handleValueChange = (_, payload) => {
        setValue(payload.value);
    };

    return (
        <React.Fragment>
            <div style={{ width: isMobile() ? '100%' : 320 }}>
                <AmountInput
                    value={value}
                    onChange={handleValueChange}
                    placeholder='Введите сумму'
                    labelView={isMobile() ? 'outer' : 'inner'}
                    label='Сумма'
                    minority={100}
                    bold={false}
                    block={true}
                    size={isMobile() ? 48 : 56}
                    breakpoint={BREAKPOINT}
                    zeroValue={zeroValue}
                />
            </div>
            <Gap size={20} />
            <Divider />
            <Gap size={20} />
            <Switch
                label='Сбрасывать 0 при фокусе'
                checked={zeroValue}
                onChange={handleZeroValueChange}
            />
        </React.Fragment>
    );
});
```
